cmake_minimum_required(VERSION 3.24)
project(FractalEconomy VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(ssw_economy src/economy.cpp src/external_trade.cpp src/authority_handoff.cpp)
target_include_directories(ssw_economy PUBLIC include)
target_include_directories(ssw_economy SYSTEM PUBLIC third_party)
if(MSVC)
  target_compile_options(ssw_economy PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(ssw_economy PRIVATE -Wall -Wextra -Werror -Wpedantic)
endif()
add_executable(FractalEconomy src/main.cpp)
target_link_libraries(FractalEconomy PRIVATE ssw_economy)
if(WIN32)
  target_link_libraries(FractalEconomy PRIVATE ws2_32 bcrypt)
endif()
add_executable(EconomyTests tests/economy_tests.cpp)
target_link_libraries(EconomyTests PRIVATE ssw_economy)
enable_testing()
add_test(NAME NativeEconomy COMMAND EconomyTests "${CMAKE_CURRENT_SOURCE_DIR}/../Content/GameData")

find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME EconomyServiceLifecycle COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/service_tests.py" "$<TARGET_FILE:FractalEconomy>" "${CMAKE_CURRENT_SOURCE_DIR}/../Content/GameData")
set_tests_properties(NativeEconomy EconomyServiceLifecycle PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE)

if(MSVC)
  target_compile_options(FractalEconomy PRIVATE /W4 /WX /permissive-)
  target_compile_options(EconomyTests PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(FractalEconomy PRIVATE -Wall -Wextra -Werror -Wpedantic)
  target_compile_options(EconomyTests PRIVATE -Wall -Wextra -Werror -Wpedantic)
endif()
